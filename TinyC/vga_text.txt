                __asm("MOV R0, [BP + -6]"); // Читаем color
                __asm("OUT R2, R0");
            }
        }
        
        // 3. Переключение цвета
        if (color == 65535) {
            color = 63488; // Красный
        } else {
            color = 65535;  // Зеленый

                // Пустой цикл
            }
        }
    }
